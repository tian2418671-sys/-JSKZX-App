```java
    private static final String KEY_ROOT_URI = "root_uri";
    private static final String KEY_ROOT_PATH = "root_path";           // P1 路径后端根路径
    private static final String KEY_ROOT_PERSISTED = "root_persisted"; // 记录最近一次SAF授权是否成功持久化

    private void setRootUri(Uri uri, boolean persisted) {
        SharedPreferences.Editor ed = prefs().edit();

              .remove(KEY_ROOT_PATH);
        }
        ed.apply();
        clearRelIndex();
    }
```

### 2.3 授权持久化与诊断回传（`tryPersistUriPermission` / `pickFolderResult` / `libraryInfo` / `pickPushFolderResult`）

```java
    /**
     * 按选择器实际返回的 flags 掩码执行持久化授权，支持只读降级。
     * @return 实际持久化的 flags（0 表示持久化失败或 Provider 未授予 PERSISTABLE 标志）
     */
    private int tryPersistUriPermission(Uri uri, Intent data) {
        if (uri == null || data == null) return 0;
        int flags = data.getFlags();
        if ((flags & Intent.FLAG_GRANT_PERSISTABLE_URI_PERMISSION) == 0) {
            android.util.Log.w("Perf", "Provider 未授予 FLAG_GRANT_PERSISTABLE_URI_PERMISSION: " + uri);
            return 0;
        }
        int takeFlags = flags & (Intent.FLAG_GRANT_READ_URI_PERMISSION | Intent.FLAG_GRANT_WRITE_URI_PERMISSION);
        if (takeFlags == 0) takeFlags = Intent.FLAG_GRANT_READ_URI_PERMISSION;
        try {
            getContext().getContentResolver().takePersistableUriPermission(uri, takeFlags);
            return takeFlags;
        } catch (SecurityException e) {
            // 若因请求 WRITE 失败，尝试降级仅持久化 READ
            if ((takeFlags & Intent.FLAG_GRANT_WRITE_URI_PERMISSION) != 0) {
                try {
                    getContext().getContentResolver().takePersistableUriPermission(
                            uri, Intent.FLAG_GRANT_READ_URI_PERMISSION);
                    return Intent.FLAG_GRANT_READ_URI_PERMISSION;
                } catch (SecurityException ex) {
                    android.util.Log.w("Perf", "只读降级持久化仍失败: " + ex.getMessage());
                }
            }
            return 0;
        }
    }

    @com.getcapacitor.annotation.ActivityCallback
    private void pickFolderResult(PluginCall call, ActivityResult result) {
```

### 2.4 Resolver 级 `EXTRA_LOADING` 异步等待（`queryChildrenSettled`）

在 `LibraryFsPlugin.java:150`（`buildRelIndex` 上方）新增以下方法，并将 `buildRelIndex()` 第 `179` 行的 `getContext().getContentResolver().query(childrenUri, DOC_CHILD_PROJECTION, null, null, null)` 替换为 `queryChildrenSettled(childrenUri, 1500L)`：

```java
    /**
     * 带 EXTRA_LOADING 等待的子项查询（采用 Resolver 级 ContentObserver）。
     * 注意：此方法会阻塞等待 Provider 异步补行，必须仅在后台线程（SCAN_EXEC / BATCH_EXEC）调用。
     */
    private android.database.Cursor queryChildrenSettled(Uri childrenUri, long timeoutMs) {
        final android.content.ContentResolver cr = getContext().getContentResolver();
        android.database.Cursor c = cr.query(childrenUri, DOC_CHILD_PROJECTION, null, null, null);
        if (c == null) return null;
        boolean loading = false;
        try {
            android.os.Bundle extras = c.getExtras();
            loading = extras != null && extras.getBoolean(DocumentsContract.EXTRA_LOADING, false);
        } catch (Exception ignored) { }
        if (!loading) return c;

        final java.util.concurrent.CountDownLatch latch = new java.util.concurrent.CountDownLatch(1);
        android.database.ContentObserver obs = new android.database.ContentObserver(
                new android.os.Handler(android.os.Looper.getMainLooper())) {
            @Override
            public void onChange(boolean selfChange) {
                latch.countDown();
            }
        };
        cr.registerContentObserver(childrenUri, true, obs);
        try {
            if (!latch.await(timeoutMs, java.util.concurrent.TimeUnit.MILLISECONDS)) {
                android.util.Log.w("Perf", "EXTRA_LOADING wait timeout: " + childrenUri);
            }
        } catch (InterruptedException ignored) {
            Thread.currentThread().interrupt();
        } finally {
            try { cr.unregisterContentObserver(obs); } catch (Exception ignored) { }
            try { c.close(); } catch (Exception ignored) { }
        }
        return cr.query(childrenUri, DOC_CHILD_PROJECTION, null, null, null);
    }
```

### 2.5 `scanFolder` 与 `scanFolderResult` 后台化改造（严格保持原返回契约）

定位到 `LibraryFsPlugin.java:1894–1965` 的 `scanFolder` 与 `scanFolderResult`，完整替换为：

```java
    @PluginMethod()
    public void scanFolder(final PluginCall call) {

    @com.getcapacitor.annotation.ActivityCallback
    private void scanFolderResult(final PluginCall call, ActivityResult result) {
        if (call == null) return;
        if (result == null || result.getResultCode() != android.app.Activity.RESULT_OK
                || result.getData() == null) {
            call.reject("用户取消选择目录");
            return;
        }
        final Intent data = result.getData();
        final Uri treeUri = data.getData();
        if (treeUri == null) {
            call.reject("未获取到目录授权");
            return;
        }
        tryPersistUriPermission(treeUri, data);

        final String ext = call.getString("ext", ".png").toLowerCase(Locale.ROOT);
        final boolean skipLarge = call.getBoolean("skipLarge", true);
        final long maxSize = 50L * 1024 * 1024;

        // 移入 SCAN_EXEC 后台线程，使用 queryChildrenSettled + 显式投影替代主线程 DocumentFile.listFiles()
        SCAN_EXEC.execute(() -> {
            try {
                DocumentFile root = DocumentFile.fromTreeUri(getContext(), treeUri);
                if (root == null || !root.exists()) {
                    call.reject("目录不存在或无法访问");
                    return;
                }
                String rootDocId = DocumentsContract.getTreeDocumentId(treeUri);
                JSArray files = new JSArray();
                java.util.ArrayDeque<String[]> stack = new java.util.ArrayDeque<>();
                stack.push(new String[]{ rootDocId, "" });

                int scannedDirs = 0;
                int foundFiles = 0;
                long lastEmit = 0L;

                while (!stack.isEmpty()) {
                    String[] cur = stack.pop();
                    String dirDocId = cur[0];
                    String prefix = cur[1];
                    scannedDirs++;

                    Uri childrenUri = DocumentsContract.buildChildDocumentsUriUsingTree(treeUri, dirDocId);
                    android.database.Cursor c = null;
                    try {
                        c = queryChildrenSettled(childrenUri, 1500L);
                        if (c == null) continue;
                        int idxId = c.getColumnIndex(DocumentsContract.Document.COLUMN_DOCUMENT_ID);
                        int idxName = c.getColumnIndex(DocumentsContract.Document.COLUMN_DISPLAY_NAME);
                        int idxMime = c.getColumnIndex(DocumentsContract.Document.COLUMN_MIME_TYPE);
                        int idxSize = c.getColumnIndex(DocumentsContract.Document.COLUMN_SIZE);

                        while (c.moveToNext()) {
                            String name = idxName >= 0 ? c.getString(idxName) : null;
                            if (name == null || name.isEmpty() || name.startsWith(".")) continue;
                            String childId = idxId >= 0 ? c.getString(idxId) : null;
                            if (childId == null) continue;
                            String mime = idxMime >= 0 ? c.getString(idxMime) : "";
                            long size = (idxSize >= 0 && !c.isNull(idxSize)) ? c.getLong(idxSize) : 0L;
                            String relPath = prefix.isEmpty() ? name : (prefix + "/" + name);

                            if (DocumentsContract.Document.MIME_TYPE_DIR.equals(mime)) {
                                if ("__trash".equalsIgnoreCase(name) || ".trash".equalsIgnoreCase(name)) continue;
                                stack.push(new String[]{ childId, relPath });
                            } else {
                                String lower = name.toLowerCase(Locale.ROOT);
                                boolean match = ext.isEmpty() || lower.endsWith(ext)
                                        || (".png".equals(ext) && lower.endsWith(".webp"));
                                if (!match) continue;
                                if (skipLarge && size > maxSize) continue;

                                JSObject item = new JSObject();
                                item.put("path", relPath);
                                item.put("name", name);
                                item.put("size", size);
                                files.put(item);
                                foundFiles++;
                            }
                        }
                    } finally {
                        if (c != null) try { c.close(); } catch (Exception ignored) { }
                    }

                    long now = System.currentTimeMillis();
                    if (now - lastEmit > 250L) {
                        lastEmit = now;
                        JSObject prog = new JSObject();
                        prog.put("dirs", scannedDirs);
                        prog.put("found", foundFiles);
                        prog.put("current", prefix);
                        notifyListeners("scanProgress", prog);
                    }
                }

                // 严格保持既有返回契约：success / treeUri / files / title
                JSObject ret = new JSObject();
                ret.put("success", true);
                ret.put("treeUri", treeUri.toString());
                ret.put("files", files);
                ret.put("title", root.getName() != null ? root.getName() : "扫描目录");
                call.resolve(ret);
            } catch (Exception e) {
                call.reject("扫描失败: " + e.getMessage());
            }
        });
    }
```

```java
                            v.put("name", sv.getDescription(getContext()));
                            v.put("path", dir.getAbsolutePath());
                            v.put("primary", false);
                            volumes.put(v);
                        }
                    }
                }
            }
        } catch (Exception ignored) { }

        JSObject ret = new JSObject();
        ret.put("granted", granted);
        ret.put("supported", true);
        ret.put("volumes", volumes);
        call.resolve(ret);
    }

    /** 跳转系统设置页申请「所有文件访问权限」，并在用户返回 App 时自动复查授权状态 */
    @PluginMethod()
    public void requestAllFilesAccess(PluginCall call) {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.R) {
            allFilesAccessState(call);
            return;
        }
        try {
            Intent intent = new Intent(android.provider.Settings.ACTION_MANAGE_APP_ALL_FILES_ACCESS_PERMISSION);
            intent.setData(Uri.parse("package:" + getContext().getPackageName()));
            startActivityForResult(call, intent, "allFilesAccessResult");
        } catch (Exception e) {
            // 魔改 ROM 不支持包名定向页时，回退到总列表页
            try {
                Intent fallback = new Intent(android.provider.Settings.ACTION_MANAGE_ALL_FILES_ACCESS_PERMISSION);
                startActivityForResult(call, fallback, "allFilesAccessResult");
            } catch (Exception ex) {
                call.reject("无法打开系统所有文件访问权限设置: " + ex.getMessage());
            }
        }
    }

    @com.getcapacitor.annotation.ActivityCallback
    private void allFilesAccessResult(PluginCall call, ActivityResult result) {
        if (call == null) return;
        allFilesAccessState(call);
    }

    /** 路径级目录浏览（目录优先、按名称排序，放入后台线程防阻塞） */
    @PluginMethod()
    public void listDirByPath(final PluginCall call) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R && !Environment.isExternalStorageManager()) {
            call.reject("未授予「所有文件访问权限」");
            return;
        }
        final String defaultPath = Environment.getExternalStorageDirectory().getAbsolutePath();
        final String path = call.getString("path", defaultPath);
        SCAN_EXEC.execute(() -> {
            File dir = new File(path);
            if (!dir.exists() || !dir.isDirectory()) {
                call.reject("目录不存在: " + path);
                return;
            }
            File[] children = dir.listFiles();
            if (children == null) {
                call.reject("目录不可读（若为 Android/data 或 Android/obb 属系统硬限制）");
                return;
            }
            java.util.Arrays.sort(children, (a, b) -> {
                if (a.isDirectory() != b.isDirectory()) return a.isDirectory() ? -1 : 1;
                return a.getName().compareToIgnoreCase(b.getName());
            });
            JSArray out = new JSArray();
            for (File f : children) {
                if (f.getName().startsWith(".")) continue;
                JSObject o = new JSObject();
                o.put("path", f.getAbsolutePath());
                o.put("name", f.getName());
                o.put("isDirectory", f.isDirectory());
                o.put("size", f.isFile() ? f.length() : 0L);
                o.put("mtime", f.lastModified());
                out.put(o);
            }
            File parent = dir.getParentFile();
            JSObject ret = new JSObject();
            ret.put("success", true);
            ret.put("path", dir.getAbsolutePath());
            ret.put("parentPath", (parent != null && parent.canRead()) ? parent.getAbsolutePath() : "");
            ret.put("files", out);
            call.resolve(ret);
        });
    }

    /** 切换为路径级库根（自动清除 KEY_ROOT_URI 并失效旧索引） */
    @PluginMethod()
    public void setLibraryRootPath(PluginCall call) {
        String path = call.getString("path", "");
        if (path == null || path.isEmpty()) {
            call.reject("路径不能为空");
            return;
        }
        File dir = new File(path);
        if (!dir.isDirectory() || !dir.canRead()) {
            call.reject("所选目录不可读");
            return;
        }
        prefs().edit()
               .putString(KEY_ROOT_PATH, dir.getAbsolutePath())
               .remove(KEY_ROOT_URI)
               .remove(KEY_ROOT_PERSISTED)
               .apply();
        clearRelIndex();
        JSObject ret = new JSObject();
        ret.put("success", true);
        ret.put("backend", "path");
        ret.put("path", dir.getAbsolutePath());
        call.resolve(ret);
    }
```
